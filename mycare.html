async function submitReview() { const r = parseInt(document.getElementById('revRating').value); if(r === 0) return alert("Please provide a star rating."); await supabaseClient.from('job_cards').update({ review_rating: r, is_reviewed: true, review_mech_feedback: document.getElementById('revMech').value, review_spares_feedback: document.getElementById('revSpares').value, review_lube_feedback: document.getElementById('revLube').value, review_advice: document.getElementById('revAdvice').value }).eq('id', pendingReviewJobId); // NOW FETCHES CO-MECHANIC DATA AS WELL const { data: jc } = await supabaseClient.from('job_cards').select('mech_id, co_mech_id, labour_charges, job_card_items(warehouse_type, total_price)').eq('id', pendingReviewJobId).single(); if (jc && (jc.mech_id || jc.co_mech_id)) { const searchId = jc.mech_id || jc.co_mech_id; // Check penalty against primary tech const today = new Date(); const day = today.getDay() || 7; if (day !== 1) today.setHours(-24 * (day - 1)); today.setHours(0,0,0,0); const startOfWeek = today.toISOString(); const { count: negativeCount } = await supabaseClient.from('job_cards').select('*', { count: 'exact', head: true }).eq('mech_id', searchId).eq('is_reviewed', true).lt('review_rating', 3).gte('closed_at', startOfWeek); let pkgLabor = 0; if(jc.job_card_items) { jc.job_card_items.forEach(item => { if(item.warehouse_type === 'Service') pkgLabor += parseFloat(item.total_price) || 0; }); } const totalLaborForJob = (parseFloat(jc.labour_charges) || 0) + pkgLabor; const baseRewardPoints = totalLaborForJob * 0.10; // 50/50 SPLIT LOGIC let actualReward = baseRewardPoints; if (jc.mech_id && jc.co_mech_id) { actualReward = baseRewardPoints / 2; } if (r === 5 && negativeCount <= 2) { if (jc.mech_id) { const { data: mech1 } = await supabaseClient.from('mechanics').select('earned_points').eq('id', jc.mech_id).single(); await supabaseClient.from('mechanics').update({ earned_points: (parseFloat(mech1.earned_points) || 0) + actualReward }).eq('id', jc.mech_id); } if (jc.co_mech_id) { const { data: mech2 } = await supabaseClient.from('mechanics').select('earned_points').eq('id', jc.co_mech_id).single(); await supabaseClient.from('mechanics').update({ earned_points: (parseFloat(mech2.earned_points) || 0) + actualReward }).eq('id', jc.co_mech_id); } } } alert("Thank you for your feedback! / आपकी प्रतिक्रिया के लिए धन्यवाद!"); closeReview(); loadDashboardData(); }
